cmake_minimum_required(VERSION 3.20)
project(vivo_connect_qt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network DBus)
add_executable(vivo-connect-qt main.cpp)
target_link_libraries(vivo-connect-qt PRIVATE Qt6::Widgets Qt6::Network Qt6::DBus)
target_compile_definitions(vivo-connect-qt PRIVATE PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
find_package(PkgConfig REQUIRED)
pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3)
add_executable(vivo-drag-fuse drag-fuse.cpp)
target_link_libraries(vivo-drag-fuse PRIVATE Qt6::Core Qt6::Network PkgConfig::FUSE3)
