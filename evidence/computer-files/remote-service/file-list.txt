const {
    Worker, 
    isMainThread, 
    parentPort, 
    workerData
} = require('worker_threads');
const pathUtils = require('path');
const fs = require('fs');
const { exec } = require('child_process');
const { platform } = require('os');

/**
 * 实现合并分片文件为完整文件
 */
if (isMainThread) {
    const __logInfo = (fun, text)=>{
        // console.log(fun, text)
        global?.__log?.info(`[fileTransfer-fileList][${fun}] ${text}`) || (()=>{})
    };

    module.exports = function fileList(originPath) {
        return new Promise(async resolve=>{
            const folderPath = pathUtils.join(originPath);

            try {
                const stats = await fs.statSync(folderPath);
                if (!stats.isDirectory()) {
                    resolve([]);
                    __logInfo('params', '不是目录')
                    return;
                }
            } catch(e) {
                __logInfo('params', `路径错误 error: ${e.message}`);
                resolve([]);
                return;
            }

            const worker = new Worker(__filename, {
                workerData: {
                    folderPath
                }
            });
            worker.on('message', (res)=>{
                const { type } = res;

                if (type === 'result') {
                    __logInfo('fileList resulte in mainworker===========>');
                    // 读取mac无权限的系统目录时， 返回结果如下, 返回空列表
                    // {"type":"result","msg":"EACCES: permission denied, scandir '/var/agentx'"}
                    if(process.platform === "darwin" && res.msg && res.msg.indexOf('permission denied' > -1))  {
                        __logInfo('fileList resulte in mainworker 2===========>', 'EACCES: permission denied');
                        resolve([])
                    }else {
                        resolve(res.result || []);
                    }
                    setTimeout(()=>{
                        worker.terminate();
                    }, 100)
                } else if (type === 'info') {
                    __logInfo('worker', res.msg);
                }
            });
            worker.on('error', (err)=>{
                console.log('==', err);
                __logInfo('worker-error', err.message);
                resolve([]);
                setTimeout(()=>{
                    worker.terminate();
                }, 100)
            });
            worker.on('exit', (code) => {
                if (code !== 0)  __logInfo('exit', code);
            });
        })
    };
} else {
    const iconv = require('iconv-lite');
    const { folderPath  } = workerData;
    const { MACDIRS, MACDICTINNARY} = require("../utils/const")
    const __log = function(msg) {
        parentPort.postMessage({
            type: 'info',
            msg
        });
    }

    async function getHiddenFiles (folderPath) {
        if(process.platform === "win32") {
            return new Promise((resolve) => {
                const cmdParms = `dir /a:h /b "${folderPath}"`
                exec(cmdParms, { encoding: 'buffer' } , (error, stdout, stderr) => {
                    if (error) {
                        const text = iconv.decode(stderr, 'gbk');
                        if (!text.includes('找不到文件')) {
                            __log(`getHiddenFiles-error: ${text}`);
                        }
                      
                        resolve({});
                        return
                    }
                    try {
                        stdout = iconv.decode(stdout, 'gbk');
                        const hiddenFilesMap = stdout.replace(/\r/g,"").split((/\r?\n/)).reduce((result, val) => {
                            if (val) {
                                result[val.trim()] = true;
                            }
                            return result;
                        }, {}) // 去除换行和空格符
                        resolve(hiddenFilesMap)
                    } catch(e) {
                        __log(`getHiddenFiles-error: ${e.message}`);
                        resolve({});
                    }
                })
            });
        }else {
            // mac为空
            return {}
        } 
    }

    async function getFileStats (path) {
        return new Promise(resolve=>{
          fs.stat(path, (err, stats) =>{
            const result = {
              status: 0,
              path,
              stats
            }
      
            if (err) {
                // __log(`getFileStats-error:${err.message}`);
                result.status = 1;
            }
            
            resolve(result);
          })
        })
      }
      // 判断是否是中文
      function isChinese(str) {
        return /^[\u4e00-\u9fa5]$/.test(str);
      } 


      //是否是英文字符
      function isEnglishPunctuation(str) {
        return /^[!@#$%^&*()_+{}\[\]:";'<>,.?\/\\|~-]$/.test(str);
      }

     
      // 判断是否是中文符号
      function isChinesePunctuation(str) {
        return /^[\u3000-\u303F\uFF00-\uFFEF\u2000-\u206F]+$/.test(str);
      }
     
      // 判断是否是数字
      function isNumber(str) {
         return /^\d+$/.test(str);
      }

      // 判断是否是英文字符
      function isEnglish(str) {
        return /^[a-zA-Z]$/.test(str);
      }
      
    __log('before fs.readdir===========>');
    fs.readdir(folderPath, async (err, files) => {
        if (err) {
            parentPort.postMessage({
                type: 'result',
                msg: err.message,
                result: []
            });
            __log(`readdir error：${err.message}`);
            return [];
        } else {
            // console.log('readdir files=======>', files);
            __log(`after fs.readdir===========>`);
            // 获取Windows的隐藏文件
            const hiddenFileMap = await getHiddenFiles(folderPath) || {}

            const fileStatsPromises = []
            let fileList = files;
            // 遍历读取到的文件列表，标记是文件 or 文件夹、以及名称、状态
            // mac 一级目录需要排除"Users"、"Library" 、 "System" 后重新排序
            // dev 目录是系统目录，文件是动态变化，常规用户不需要访问这个文件， 也无法读取， 给过滤掉
            if(process.platform === "darwin" && folderPath == "/") {
                fileList = MACDIRS.concat(
                    files.filter((name) => {
                        return  name != "Users" && name != "Library" && name != "System" && name != "Applications" && name != "dev"
                    })
                )
            }
            fileList.forEach(async fileName => {
                // win过滤掉的文件（夹），主要是隐藏文件， mac 不需要过滤
                if (process.platform === "win32") {
                    if (hiddenFileMap[fileName]) {
                        return;
                    }
                    if (pathUtils.extname(fileName) === '.lnk') {
                        return;
                    }
                    if (fileName.startsWith('.')) {
                        return;
                    } // 不是.开头
                    if (fileName.startsWith('_')) {
                        return;
                    } // 不是_开头
                    if (fileName.startsWith('~$')) {
                        return
                    }; // 不是~$开头
                }
                // 读取文件信息
                fileStatsPromises.push(new Promise(async (resolve) => {
                    const filePath = pathUtils.join(folderPath, fileName);
                    const res = await getFileStats(filePath);
                    res.fileName = fileName;
                    resolve(res);
                }))
            });
            const res = await Promise.all(fileStatsPromises);
            __log('fileStats all done===========>');
            // console.log('res in fileList readdir=========>', res)
            // 过滤掉异常文件和符号链接， 添加文件或文件type、size、time
            const result = res.filter(fileStats => {
                const { status, stats } = fileStats;
                // 获取文件信息正常 && 是文件或目录 && 不是符号链接
                return (status === 0)  && (stats.isFile() || stats.isDirectory()) && !stats.isSymbolicLink(); 
            }).map(fileStats => {
                const { fileName, stats} = fileStats;
                // console.log('stats.isFile()========>',fileName,  stats.isFile())
                const type = stats.isFile() ? 'file' : 'folder'
                // 文件修改时间 和创建时间
                const { mtimeMs, birthtimeMs } = stats 
                const time = new Date(mtimeMs || birthtimeMs).getTime();
                // 文件夹的size返回0
                const size = (type == "file") ? stats.size : 0;
                const name = fileName;
                // mac需要获取获取中文名称
                const ChineseName = (process.platform === "darwin" && process.env.REGION != "foreign") ? (MACDICTINNARY[name] || "") : ""
                return {
                    type,
                    name,
                    ChineseName,
                    time,
                    size
                }
            });

            parentPort.postMessage({
                type: 'result',
                result
            });
        }
    })
}
