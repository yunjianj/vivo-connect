const { getRootLocalStorage , setRootLocalStorage, getLocalStorage, getMessageId, getDeviceNo } = require('../utils/common.js');
function startHeartCheck(pingTime) {
    global.__log.info('开始心跳检测========================》', new Date());
    clearTimer();
    global.pingInterval = setInterval(() => {
        const wsParams  = productWsParam({action: "ping"});
        global.WS.handlePing(wsParams);
        // 2s 后检查pong
        // global.pongTimeOut = setTimeout(() => {
        //     global.WS.checkPong();
        // }, 2000);
    }, pingTime);
}


/**
 * 根据传参生成websocket消息
 * @param {*} params {对象}
 * @returns 
 */
function productWsParam( params) {
    const { action, param, msgId} = params;
    const { connectionId = "", targetDeviceNo = "" } = getRootLocalStorage("websocketConnectionData") || {};
    // global.__log.info(`获取参数息数据========================》${JSON.stringify(getRootLocalStorage("websocketConnectionData"))}`);

    const wsParams = {
        "msgId": msgId || getMessageId(),
        "timeStamp": Date.now(),
        "action": action || "msg",
        "sourceDeviceNo": getDeviceNo(),
        "targetDeviceNo": targetDeviceNo,
        "data": param || "", 
        "version":"v1",
        "connectionId": connectionId, 
    }
    // global.__log.info(`websocket 发送消息数据========================》${JSON.stringify(wsParams)}`);

    return wsParams
}



const getDeviceInfo = () =>{
    // console.log('deviceInfo====>', deviceInfo);
    const deviceInfo = getLocalStorage("deviceInfo");
    return deviceInfo
}


function clearTimer() {
    // clearTimeout(global.pongTimeOut);
    clearInterval(global.pingInterval);
}

/**
 *  设置更新缓存在本机的电脑账户密码校验情况， 防爆破机制
 * @param {*} targetDeviceNo  移动端的设备代码
 * @param {*} type  更新（update）、重置（reset， 重置时counter置为0）
*/ 
function updatePasswordVerification(targetDeviceNo, type) {
    let passwordVerificationData = getRootLocalStorage("passwordVerificationData") || {};
    console.log(" passwordVerificationData in init=========>", JSON.stringify(passwordVerificationData), type)
    let count = 0
    if(type == "update") {
        const lastCount = passwordVerificationData[targetDeviceNo] && passwordVerificationData[targetDeviceNo]["count"] || 0;
        // 密码校验次数在原来的基础上增加一次
        count = lastCount ? lastCount + 1 : 1;
    }
    passwordVerificationData[targetDeviceNo] = {
        count,
        timeStamp: Date.now()
    }
    global.__log.info(` data in updatePasswordVerification========> ${JSON.stringify(passwordVerificationData)}`);
    setRootLocalStorage("passwordVerificationData", passwordVerificationData)
    return passwordVerificationData[targetDeviceNo]
}

/**
 *  文件传输场景下，检测当前移动端访问pc 是否被锁定分钟， 限制访问。
 * @param {*} targetDeviceNo  移动端的设备代码
 * @returns {count, timeStamp} or false
*/ 
function checkAcessLock(targetDeviceNo) {
    if(global.connectionType == 0) {
        return false
    }
    let passwordVerificationData = getRootLocalStorage("passwordVerificationData")|| {};
    const targetData = passwordVerificationData[targetDeviceNo];
    console.log('targetData=========>', targetData)
    if(!targetData) {
        return false
    }
    const { count, timeStamp } = targetData;
    const nowDate = Date.now();
    const timeGap = nowDate - timeStamp
    console.log('timeGap=========>',nowDate,  timeGap)
    // 当前时间 - 记录时间戳小于3分钟， 被判断为锁定
    if(timeGap < 3 * 60 * 1000) {
        return targetData
    // 大于3分钟了重置，剩下
    }else{
        updatePasswordVerification(targetDeviceNo, "reset");
        return false
    }
}
